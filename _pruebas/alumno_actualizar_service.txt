  async actualizar(id, datos, usuarioLogueado) {
    const existente = await this.#alumnos.obtenerPorId(id);
    if (!existente) throw new Error("Estudiante no encontrado");
    if (usuarioLogueado.rol === "coordinador") {
      if (String(existente.establecimiento.id) !== String(usuarioLogueado.establecimiento?._id)) {
        throw new Error("El estudiante no pertenece a su establecimiento");
      }
    } else if (usuarioLogueado.rol !== "admin") {
      throw new Error("No tiene permisos para editar estudiantes");
    }
    // Validar con dominio
    if (datos.nombre || datos.rut) {
      new Alumno(
        datos.rut || existente.rut,
        datos.nombre || existente.nombre,
        existente.fechaNacimiento,
        existente.apoderado,
        existente.email,
        existente.telefono,
        datos.genero || existente.genero
      );
    }
    const actualizar = {};
    if (datos.rut) actualizar.rut = datos.rut;
    if (datos.nombre) actualizar.nombre = datos.nombre;
    if (datos.genero) actualizar.genero = datos.genero;
    if (datos.fechaNacimiento) actualizar.fechaNacimiento = datos.fechaNacimiento;
    if (datos.apoderado !== undefined) actualizar.apoderado = datos.apoderado;
    if (datos.email !== undefined) actualizar.email = datos.email;
    if (datos.telefono !== undefined) actualizar.telefono = datos.telefono;
    await this.#alumnos.actualizar(id, actualizar);
    return this.#aPublico ? this.#aPublico(await this.#alumnos.obtenerPorId(id)) : await this.#alumnos.obtenerPorId(id);
  }
